function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); initSheets(sheet); var callback = e ? e.parameter.callback : null; var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "getRestaurants"; var result = []; try { // 1. 取得餐廳清單 if (action === "getRestaurants") { var resSheet = sheet.getSheetByName("Restaurants"); var data = resSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0] && data[i][0].toString().trim() !== "") { result.push({ name: String(data[i][0]), phone: String(data[i][1] || ""), address: String(data[i][2] || ""), tags: data[i][3] ? String(data[i][3]).split(",").filter(Boolean) : [], note: String(data[i][4] || "") }); } } } // 2. 取得菜單 else if (action === "getMenu") { var resName = e.parameter.restaurant; var menuSheet = sheet.getSheetByName("Menu"); var data = menuSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === resName) { result.push({ item: String(data[i][1]), price: String(data[i][2]) }); } } } // 3. 取得點餐明細 else if (action === "getOrders") { var restaurantName = e.parameter.restaurant; var orderSheet = sheet.getSheetByName("Orders"); var data = orderSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0] && data[i][0].toString().trim() !== "") { if (!restaurantName || String(data[i][0]) === restaurantName) { result.push({ restaurant: String(data[i][0]), user: String(data[i][1]), item: String(data[i][2]), price: String(data[i][3]), note: String(data[i][4] || "") }); } } } } // 4. 取得雲端鎖定餐廳資訊 (全設備通用) else if (action === "getPinned") { var setSheet = sheet.getSheetByName("Settings"); var data = setSheet.getDataRange().getValues(); var pinnedStr = (data.length > 1 && data[1][0]) ? String(data[1][0]) : ""; var dateStr = (data.length > 1 && data[1][1]) ? String(data[1][1]) : ""; result = { pinned: pinnedStr ? pinnedStr.split(",").filter(Boolean) : [], dateStr: dateStr }; } // 5. 儲存雲端鎖定餐廳資訊 else if (action === "savePinned") { var setSheet = sheet.getSheetByName("Settings"); var pinnedStr = e.parameter.pinned || ""; var dateStr = e.parameter.dateStr || ""; setSheet.getRange("A2:B2").setValues([[pinnedStr, dateStr]]); result = { status: "SUCCESS" }; } // 6. 新增餐廳 else if (action === "addRestaurant") { var resSheet = sheet.getSheetByName("Restaurants"); resSheet.appendRow([ e.parameter.name || "", e.parameter.phone || "", e.parameter.address || "", e.parameter.tags || "", e.parameter.note || "" ]); result = { status: "SUCCESS" }; } // 7. 新增菜單單項 else if (action === "addMenuItem") { var menuSheet = sheet.getSheetByName("Menu"); menuSheet.appendRow([ e.parameter.restaurant || "", e.parameter.item || "", e.parameter.price || "0" ]); result = { status: "SUCCESS" }; } // 8. 刪除餐廳及其相關資料 else if (action === "deleteRestaurant") { var name = e.parameter.name; var resSheet = sheet.getSheetByName("Restaurants"); var rows = resSheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === name) { resSheet.deleteRow(i + 1); break; } } var menuSheet = sheet.getSheetByName("Menu"); var menuRows = menuSheet.getDataRange().getValues(); for (var k = menuRows.length - 1; k >= 1; k--) { if (String(menuRows[k][0]) === name) { menuSheet.deleteRow(k + 1); } } result = { status: "SUCCESS" }; } // 9. 新增點餐 else if (action === "addOrder") { var orderSheet = sheet.getSheetByName("Orders"); orderSheet.appendRow([ e.parameter.restaurant || "", e.parameter.user || "", e.parameter.item || "", e.parameter.price || "0", e.parameter.note || "" ]); result = { status: "SUCCESS" }; } // 10. 刪除點餐 else if (action === "deleteOrder") { var orderSheet = sheet.getSheetByName("Orders"); var rows = orderSheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === e.parameter.restaurant && String(rows[i][1]) === e.parameter.user && String(rows[i][2]) === e.parameter.item && String(rows[i][4] || "") === (e.parameter.note || "")) { orderSheet.deleteRow(i + 1); break; } } result = { status: "SUCCESS" }; } } catch(err) { result = { status: "ERROR", message: err.toString() }; } var jsonResponse = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + jsonResponse + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonResponse) .setMimeType(ContentService.MimeType.JSON); } } function initSheets(sheet) { var resSheet = sheet.getSheetByName("Restaurants") || sheet.insertSheet("Restaurants"); if (resSheet.getLastRow() === 0) { resSheet.appendRow(["餐廳名稱", "電話", "地址", "分類標籤", "備註/菜單說明"]); } var menuSheet = sheet.getSheetByName("Menu") || sheet.insertSheet("Menu"); if (menuSheet.getLastRow() === 0) { menuSheet.appendRow(["餐廳名稱", "餐點名稱", "價格"]); } var orderSheet = sheet.getSheetByName("Orders") || sheet.insertSheet("Orders"); if (orderSheet.getLastRow() === 0) { orderSheet.appendRow(["餐廳名稱", "點餐人", "餐點品項", "金額", "客製備註"]); } var setSheet = sheet.getSheetByName("Settings") || sheet.insertSheet("Settings"); if (setSheet.getLastRow() === 0) { setSheet.appendRow(["PinnedRestaurants", "PinnedDate"]); } }
